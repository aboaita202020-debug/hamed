## 🚀 الاستخدام

### من Python

```python
from ai import AdvancedAIBrain

# تهيئة النظام
ai_brain = AdvancedAIBrain()

# توليد أفكار جديدة
client_context = {
    'type': 'business',
    'industry': 'ecommerce',
    'budget': 'medium',
    'goals': ['increase_sales', 'improve_marketing']
}

ideas = ai_brain.think_and_generate_ideas(client_context)
print(f"تم توليد {len(ideas)} أفكار")

# التفاوض الذاتي
negotiation = ai_brain.negotiate_autonomously(
    client_id='client_123',
    service_type='website_analysis',
    client_budget=100,
    client_requirements={'complexity': 'medium'}
)
print(f"السعر النهائي: ${negotiation['final_price']}")

# اتخاذ قرار مستقل
decision = ai_brain.make_autonomous_decision({
    'type': 'pricing',
    'urgency': 'high',
    'risk': 'low',
    'value': 2000
})
print(f"القرار: {decision['action']}")

# تقديم عرض مخصص
proposal = ai_brain.create_proposal(
    client_id='client_456',
    service_type='seo_optimization',
    client_requirements={
        'complexity': 'high',
        'budget': 'high',
        'timeline': 'urgent'
    }
)
print(f"تم إنشاء عرض بسعر ${proposal['price']}")
```
